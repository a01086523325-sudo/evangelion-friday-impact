Psych Engine 0.7.x 소스에서 직접 빌드하는 경우 (메뉴 스크립트 기능이 없는 버전):
source/states/MainMenuState.hx 의 update() 안, 맨 위쪽에 아래 코드를 붙여 넣고
상단에 `var kaoruTyped:String = '';` 필드를 추가하면 같은 효과를 냅니다.

    var k = FlxG.keys.firstJustPressed();
    if (k >= 0)
    {
        var d = -1;
        if (k >= 48 && k <= 57) d = k - 48;
        else if (k >= 96 && k <= 105) d = k - 96;
        if (d < 0) kaoruTyped = '';
        else
        {
            kaoruTyped += d;
            if (kaoruTyped.length > 8) kaoruTyped = kaoruTyped.substr(kaoruTyped.length - 8);
            if (StringTools.endsWith(kaoruTyped, '0913') || StringTools.endsWith(kaoruTyped, '20000913'))
            {
                kaoruTyped = '';
                FlxG.sound.playMusic(Paths.music('kaoruBirthday'), 0.9, false);
                FlxG.sound.music.onComplete = function() { FlxG.sound.playMusic(Paths.music('freakyMenu'), 0.7, true); };
            }
        }
    }
